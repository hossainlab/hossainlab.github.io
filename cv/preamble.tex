\usepackage{marvosym} % Allows the use of symbols

\usepackage{enumitem} % List customization (nosep, topsep, leftmargin)

% Justified text, no hyphenation, no loose word spacing.
% Requires LuaLaTeX: microtype font expansion (unavailable in XeTeX) micro-scales
% glyphs so lines justify without stretching the interword spaces.

% (Quarto's default template already loads microtype, so configure it here
% rather than re-loading it with options, which would be an option clash.)
\microtypesetup{
  protrusion = true,
  expansion  = true,
  tracking   = true,
  factor     = 1100,  % protrusion strength
  stretch    = 25,    % max glyph expansion (per mille)
  shrink     = 25     % max glyph shrink
}

\hyphenpenalty=10000  % No hyphenation
\exhyphenpenalty=10000
\tolerance=3000       % Accept slightly looser lines rather than overfull boxes
\emergencystretch=1em % Last-resort stretch on the final pass only
\hbadness=10000       % Silence badness warnings


% Main font

\usepackage{fontspec}
\setmainfont{Inter_18pt-Regular}[
  Path           = ./fonts/,
  Extension      = .ttf,
  UprightFont    = *,
  BoldFont       = Inter_18pt-Bold,
  ItalicFont     = Inter_18pt-Italic,
  BoldItalicFont = Inter_18pt-BoldItalic
]
\setsansfont{Inter_18pt-Regular}[
  Path           = ./fonts/,
  Extension      = .ttf,
  UprightFont    = *,
  BoldFont       = Inter_18pt-Bold,
  ItalicFont     = Inter_18pt-Italic,
  BoldItalicFont = Inter_18pt-BoldItalic
]
